localStorage & sessionStorage

// localStorage — persists after browser close
localStorage.setItem("key", "value");
localStorage.getItem("key");        // "value" or null
localStorage.removeItem("key");
localStorage.clear();               // remove ALL items
localStorage.length;                 // number of items
localStorage.key(0);               // get key at index

// Store objects (must stringify)
localStorage.setItem("user", JSON.stringify({ name: "Alice", age: 30 }));
const user = JSON.parse(localStorage.getItem("user"));

// sessionStorage — cleared when tab closes
sessionStorage.setItem("token", "abc123");
sessionStorage.getItem("token");

// storage event — fires in OTHER tabs
window.addEventListener("storage", (e) => {
  console.log(e.key, e.oldValue, e.newValue);
});

JSON

// Stringify — JS to JSON string
JSON.stringify({ a: 1, b: [2, 3] });
// '{"a":1,"b":[2,3]}'

JSON.stringify(obj, null, 2);      // pretty print (2-space indent)
JSON.stringify(obj, ["name", "age"]); // only include these keys
JSON.stringify(obj, (key, val) => {
  if (key === "password") return undefined; // exclude
  return val;
});

// Parse — JSON string to JS
JSON.parse('{"a":1}');           // { a: 1 }
JSON.parse(str, (key, val) => {
  if (key === "date") return new Date(val); // reviver
  return val;
});

// Deep clone (simple objects, no functions/dates)
const clone = JSON.parse(JSON.stringify(obj));

// structuredClone — modern deep clone (handles Date, Map, etc.)
const clone2 = structuredClone(obj);

URL & URLSearchParams

const url = new URL("https://example.com/path?a=1&b=2#section");
url.protocol;     // "https:"
url.hostname;     // "example.com"
url.pathname;     // "/path"
url.search;       // "?a=1&b=2"
url.hash;         // "#section"
url.href;         // full URL string

const params = url.searchParams;
params.get("a");         // "1"
params.set("c", "3");   // add/update
params.append("a", "5");// add another a
params.delete("b");      // remove
params.has("a");         // true
params.toString();       // "a=1&a=5&c=3"
for (const [k, v] of params) console.log(k, v);

// Build URL programmatically
const base = new URL("https://api.example.com/search");
base.searchParams.set("q", "javascript");
base.searchParams.set("page", "1");
fetch(base.toString());

  (position) => {
    position.coords.latitude;
    position.coords.longitude;
    position.coords.accuracy;    // meters

    position.coords.speed;

IntersectionObserver

// Detect when elements enter/leave viewport
const observer = new IntersectionObserver((entries) => {
  entries.forEach((entry) => {
    if (entry.isIntersecting) {
      entry.target.classList.add("visible");
      observer.unobserve(entry.target);  // stop after first
    }
    entry.intersectionRatio;    // 0.0 to 1.0
    entry.boundingClientRect;
  });
}, {
  root: null,           // null = viewport
  rootMargin: "0px",
  threshold: [0, 0.5, 1],  // fire at 0%, 50%, 100%
});

document.querySelectorAll(".animate").forEach((el) => observer.observe(el));

ResizeObserver & MutationObserver

// ResizeObserver — detect size changes
const ro = new ResizeObserver((entries) => {
  entries.forEach(({ contentRect }) => {
    console.log(contentRect.width, contentRect.height);
  });
});
ro.observe(element);
ro.unobserve(element);
ro.disconnect();

// MutationObserver — detect DOM changes
const mo = new MutationObserver((mutations) => {
  mutations.forEach((mutation) => {
    mutation.type;              // "childList" | "attributes" | "characterData"
    mutation.addedNodes;
    mutation.removedNodes;
    mutation.attributeName;
  });
});
mo.observe(element, {
  childList: true,      // watch children added/removed
  subtree: true,        // include all descendants
  attributes: true,     // watch attribute changes
  characterData: true,  // watch text content
});
mo.disconnect();
