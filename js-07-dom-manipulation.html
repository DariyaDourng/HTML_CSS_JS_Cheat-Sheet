DOM Manipulation

Select, create, modify, and remove DOM elements using JavaScript.

Selecting Elements

// Single element (first match)
document.querySelector(".btn");       // any CSS selector
document.querySelector("#main");
document.querySelector("ul > li:first-child");

// All matching elements (NodeList)
document.querySelectorAll(".card");   // returns NodeList
[...document.querySelectorAll("p")].forEach(p => ...);

// Legacy methods
document.getElementById("header");
document.getElementsByClassName("active");  // HTMLCollection (live)
document.getElementsByTagName("div");

// Traverse the DOM
el.parentElement;
el.children;                  // HTMLCollection of child elements
el.firstElementChild;
el.lastElementChild;
el.nextElementSibling;
el.previousElementSibling;
el.closest(".container");     // nearest ancestor matching selector
el.matches(".active");        // true if el matches selector

Reading & Writing Content

// Text content
el.textContent;           // all text (safe, no HTML parsing)
el.textContent = "Hello"; // sets text (HTML escaped)

// HTML content (XSS risk with user input!)
el.innerHTML;
el.innerHTML = "<strong>Bold</strong>";
el.outerHTML;             // includes the element itself

// Safer alternative
el.insertAdjacentHTML("beforeend", "<span>safe-ish</span>");
// positions: "beforebegin" | "afterbegin" | "beforeend" | "afterend"

// Form values
input.value;
input.value = "new value";
checkbox.checked;
select.value;

Attributes

el.getAttribute("href");
el.setAttribute("href", "https://example.com");
el.removeAttribute("disabled");
el.hasAttribute("hidden");     // true/false

// data-* attributes
el.dataset.userId;            // reads data-user-id
el.dataset.theme = "dark";   // sets data-theme

// Properties vs attributes
input.value;                  // current value (property)
input.getAttribute("value"); // initial/default value (attribute)

CSS Classes & Styles

// classList API (recommended)
el.classList.add("active", "highlight");
el.classList.remove("active");
el.classList.toggle("open");
el.classList.toggle("open", true);   // force add
el.classList.replace("old", "new");
el.classList.contains("active");     // true/false

// Inline styles
el.style.color = "red";
el.style.backgroundColor = "#fff";  // camelCase!
el.style.cssText = "color:red; font-size:16px";
el.style.removeProperty("color");

// Get computed styles
getComputedStyle(el).color;
getComputedStyle(el).getPropertyValue("--my-var"); // CSS var

// Set CSS variable
el.style.setProperty("--color", "blue");
document.documentElement.style.setProperty("--primary", "red");

Creating & Inserting Elements

// Create
const div = document.createElement("div");
div.className = "card";
div.textContent = "Hello";

// Append (move if already in DOM)
parent.appendChild(div);
parent.append(div, "text", el2);  // accepts multiple + strings
parent.prepend(div);               // insert at beginning

// Insert relative to element
el.before(newEl);   // before el
el.after(newEl);    // after el

// insertBefore(new, reference)
parent.insertBefore(newEl, referenceEl);

// Clone
const clone = el.cloneNode(true);  // true = deep clone (with children)

// Template element
const tpl = document.getElementById("my-template");
const content = tpl.content.cloneNode(true);
document.body.appendChild(content);

Removing Elements

el.remove();                    // remove from DOM
parent.removeChild(child);      // legacy
el.replaceWith(newEl);           // replace with another
el.replaceChildren();           // remove all children
el.innerHTML = "";             // clear children (faster)

Dimensions & Position

el.getBoundingClientRect();   // {top, left, width, height, right, bottom}
el.offsetWidth; el.offsetHeight;
el.clientWidth; el.clientHeight; // without scrollbar
el.scrollWidth; el.scrollHeight;
el.scrollTop;  el.scrollLeft;
el.scrollIntoView({ behavior: "smooth", block: "start" });
