Document Structure & DOCTYPE

The skeleton of every HTML page — from DOCTYPE declaration to closing tag.

The Minimal HTML5 Document

<!-- DOCTYPE tells browser this is HTML5 -->
<!DOCTYPE html>
<html lang="en">

  <head>
    <!-- Character encoding (always first) -->
    <meta charset="UTF-8"/>

    <!-- Responsive viewport -->
    <meta name="viewport" content="width=device-width, initial-scale=1.0"/>

    <!-- Page title (shown in tab) -->
    <title>My Page</title>
  </head>

  <body>
    <!-- All visible content goes here -->
    <h1>Hello, World!</h1>
  </body>

</html>

Essential <head> Meta Tags

<head>
  <!-- 1. Character set -->
  <meta charset="UTF-8"/>

  <!-- 2. IE compatibility (legacy) -->
  <meta http-equiv="X-UA-Compatible" content="IE=edge"/>

  <!-- 3. Viewport for mobile -->
  <meta name="viewport" content="width=device-width, initial-scale=1.0"/>

  <!-- 4. SEO description -->
  <meta name="description" content="Page description for search engines"/>

  <!-- 5. Keywords (less important now) -->
  <meta name="keywords" content="html, css, javascript"/>

  <!-- 6. Author -->
  <meta name="author" content="Your Name"/>

  <!-- 7. Open Graph (social sharing) -->
  <meta property="og:title" content="My Page"/>
  <meta property="og:description" content="Description"/>
  <meta property="og:image" content="https://example.com/image.png"/>

  <!-- 8. Favicon -->
  <link rel="icon" href="favicon.ico"/>
  <link rel="icon" href="icon.svg" type="image/svg+xml"/>

  <!-- 9. CSS link -->
  <link rel="stylesheet" href="styles.css"/>

  <!-- 10. JS (defer = run after HTML parsed) -->
  <script src="app.js" defer></script>
</head>
