CSS Variables & Functions

Custom properties (CSS variables), calc(), math functions, and more.

CSS Custom Properties (Variables)

/* Define variables on :root (global scope) */
:root {
  --color-primary: #00ffd1;
  --color-bg: #0a0a0f;
  --font-size-base: 1rem;
  --spacing-md: 16px;
  --border-radius: 8px;
  --transition: 0.2s ease;
}

/* Use variables with var() */
.btn {
  background: var(--color-primary);
  padding: var(--spacing-md);
  border-radius: var(--border-radius);
  transition: all var(--transition);
}

/* var() with fallback */
.el {
  color: var(--color-accent, red);   /* use red if --color-accent not defined */
  font-size: var(--size, var(--font-size-base, 1rem));  /* nested fallback */
}

/* Scoped variables (override per component) */
.dark-card {
  --color-bg: #1a1a2e;  /* overrides root for this element and children */
  background: var(--color-bg);
}

/* Dynamic theming with data attributes */
[data-theme="light"] {
  --color-bg: #ffffff;
  --color-text: #111111;
}
[data-theme="dark"] {
  --color-bg: #0a0a0f;
  --color-text: #e8e8f0;
}

calc()

.el {
  width: calc(100% - 32px);
  height: calc(100vh - 60px);
  margin: calc(var(--spacing-md) * 2);
  font-size: calc(1rem + 0.5vw);
  padding: calc(8px + 1vw);
  /* Supports: + - * / */
  /* Mixed units are allowed */
}

Math Functions

.el {
  /* clamp(min, preferred, max) */
  font-size: clamp(1rem, 2.5vw, 2rem);
  width: clamp(280px, 50%, 800px);

  /* min() — smallest value wins */
  width: min(90%, 1200px);
  padding: min(5vw, 40px);

  /* max() — largest value wins */
  font-size: max(14px, 1vw);
  padding: max(8px, 2vw);

  /* round() — modern */
  width: round(33.3333%, 1px);

  /* abs(), sign() — modern */
  margin: abs(var(--offset));
}

counter() and counters()

ol.custom {
  counter-reset: my-counter;
  list-style: none;
}
ol.custom li {
  counter-increment: my-counter;
}
ol.custom li::before {
  content: counter(my-counter) ". ";
  color: var(--color-primary);
  font-weight: bold;
}

/* Nested counters */
ol {
  counter-reset: section;
}
li::before {
  content: counters(section, ".") ". ";
  counter-increment: section;
}

env() — Environment Variables

/* Safe area insets for notched phones */
.fixed-nav {
  padding-top: env(safe-area-inset-top);
  padding-bottom: env(safe-area-inset-bottom);
  padding-left: env(safe-area-inset-left);
  padding-right: env(safe-area-inset-right);
}

/* With fallback */
.el { padding: env(safe-area-inset-top, 20px); }

attr() — Read HTML Attributes

/* Use element's attribute value in CSS */
a::after {
  content: " (" attr(href) ")";  /* shows URL after link text */
}
[data-tooltip]::after {
  content: attr(data-tooltip);
}

@layer — Cascade Layers

/* Declare layer order (lowest to highest priority) */
@layer base, components, utilities;

@layer base {
  * { box-sizing: border-box; }
  body { margin: 0; }
}

@layer components {
  .btn { padding: 8px 16px; }
}

@layer utilities {
  .mt-4 { margin-top: 1rem; }
}
/* utilities beats components beats base, regardless of specificity */
