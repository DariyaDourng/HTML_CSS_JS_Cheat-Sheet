Operators

Every JavaScript operator — arithmetic, comparison, logical, assignment, and modern operators.

Arithmetic Operators

10 + 3   // 13  — addition
10 - 3   // 7   — subtraction
10 * 3   // 30  — multiplication
10 / 3   // 3.333... — division
10 % 3   // 1   — modulus (remainder)
2 ** 10  // 1024 — exponentiation (ES2016)

// Increment / decrement
let a = 5;
a++;      // returns 5, then a becomes 6 (post-increment)
++a;      // a becomes 7, returns 7 (pre-increment)
a--;      // returns 7, then a becomes 6
--a;      // a becomes 5, returns 5

// String concatenation with +
"Hello" + " " + "World"  // "Hello World"
"Value: " + 42           // "Value: 42" (number coerced to string)

Comparison Operators

5 ==  "5"    // true  — loose equality (coerces types)
5 === "5"    // false — strict equality (no coercion)
5 !=  "5"    // false — loose inequality
5 !== "5"    // true  — strict inequality ← prefer this

5 >  3       // true
5 <  3       // false
5 >= 5       // true
5 <= 4       // false

// Tricky equality cases
null == undefined   // true  (special case)
null === undefined  // false
NaN === NaN         // false (NaN is never equal to anything)
isNaN(NaN)          // true
Number.isNaN(NaN)   // true (stricter)

Logical Operators

true && true   // true  — AND
true && false  // false
false || true  // true  — OR
!true          // false — NOT
!!"hello"       // true  — double NOT = coerce to boolean

// Short-circuit evaluation
const a = null;
const b = a && a.name;   // null — stops at a (falsy)
const c = a || "default"; // "default" — a is falsy, uses right side
const d = "" || "fallback"; // "fallback" — "" is falsy

// Practical patterns
const name = user && user.name;         // safe property access
const display = name || "Anonymous";   // default value

Nullish Coalescing & Optional Chaining

// ?? — nullish coalescing (ES2020)
// Returns right side only if left is null or undefined
const a = null ?? "default";    // "default"
const b = 0    ?? "default";    // 0  ← unlike ||, 0 is NOT replaced
const c = ""   ?? "default";    // ""  ← empty string kept

// ??= — nullish assignment
let x = null;
x ??= "value";  // x becomes "value" (only assigns if null/undefined)

// ?. — optional chaining (ES2020)
const user = null;
user?.name;         // undefined (no error)
user?.address?.city; // undefined
user?.greet();      // undefined (no error)
arr?.[0];           // safe array access

Assignment Operators

let x = 10;
x += 5;   // x = x + 5  → 15
x -= 3;   // x = x - 3  → 12
x *= 2;   // x = x * 2  → 24
x /= 4;   // x = x / 4  → 6
x %= 4;   // x = x % 4  → 2
x **= 3;  // x = x ** 3 → 8
x &&= 5; // x = x && 5  → 5 (logical AND assign)
x ||= 9; // x = x || 9  → 5 (x is truthy, no change)
x ??= 9; // x = x ?? 9  → 5 (x not null/undefined)

Bitwise Operators

5 &  3   // 1  — AND
5 |  3   // 7  — OR
5 ^  3   // 6  — XOR
~5       // -6 — NOT
5 << 1   // 10 — left shift
5 >> 1   // 2  — right shift
5 >>> 1  // 2  — unsigned right shift

Other Operators

// Ternary (conditional)
const result = condition ? "yes" : "no";
const label = score >= 60 ? "Pass" : "Fail";

// Comma operator (rarely used)
let a = (1, 2, 3); // a = 3 (last value)

// void operator
void 0;   // undefined

// in operator — check property in object
"name" in { name: "Alice" };  // true

// instanceof — check prototype chain
[] instanceof Array;    // true
[] instanceof Object;   // true

// delete — remove object property
const obj = { a: 1, b: 2 };
delete obj.a;   // { b: 2 }

// Spread operator
const arr = [1, 2, ...[3, 4]];  // [1, 2, 3, 4]
const obj2 = { ...obj, c: 3 };   // { b: 2, c: 3 }
