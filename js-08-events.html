Events

addEventListener, all event types, bubbling, delegation, and custom events.

addEventListener

// Basic usage
el.addEventListener("click", function(event) {
  console.log(event.target);
});

// Arrow function
btn.addEventListener("click", (e) => {
  e.preventDefault();    // stop default browser action
  e.stopPropagation();   // stop bubbling up
});

// With options
el.addEventListener("click", handler, {
  once: true,       // fire only once then auto-remove
  capture: false,   // bubble phase (default)
  passive: true,    // will not call preventDefault (scroll perf)
});

// Remove listener (needs same function reference)
const handler = () => console.log("clicked");
el.addEventListener("click", handler);
el.removeEventListener("click", handler);

The Event Object

el.addEventListener("click", (e) => {
  e.type;             // "click"
  e.target;           // element that was clicked
  e.currentTarget;    // element listener is on
  e.timeStamp;        // ms since page load
  e.bubbles;          // true if event bubbles
  e.cancelable;       // true if preventDefault works
  e.preventDefault(); // stop default (link, form submit)
  e.stopPropagation();// stop bubbling
  e.stopImmediatePropagation(); // stop other listeners on same el
});

Mouse Events

el.addEventListener("click", handler);
el.addEventListener("dblclick", handler);
el.addEventListener("mousedown", handler);
el.addEventListener("mouseup", handler);
el.addEventListener("mousemove", handler);
el.addEventListener("mouseenter", handler); // doesn't bubble
el.addEventListener("mouseleave", handler); // doesn't bubble
el.addEventListener("mouseover", handler);  // bubbles
el.addEventListener("mouseout", handler);   // bubbles
el.addEventListener("contextmenu", handler); // right-click
el.addEventListener("wheel", handler);       // scroll wheel

// Mouse event properties
e.clientX; e.clientY;  // position relative to viewport
e.pageX; e.pageY;      // position relative to page
e.offsetX; e.offsetY;  // position relative to element
e.button;              // 0=left, 1=middle, 2=right
e.buttons;             // bitmask of pressed buttons
e.shiftKey; e.ctrlKey; e.altKey; e.metaKey;

Keyboard Events

document.addEventListener("keydown", (e) => {
  e.key;        // "Enter", "a", "ArrowUp", " ", "Escape"
  e.code;       // "KeyA", "Enter", "Space" (physical key)
  e.keyCode;    // deprecated number (avoid)
  e.ctrlKey;    // true if Ctrl held
  e.shiftKey;
  e.altKey;
  e.metaKey;    // Cmd on Mac
  e.repeat;     // true if key held down

  // Common patterns
  if (e.key === "Enter") submit();
  if (e.key === "Escape") closeModal();
  if (e.ctrlKey && e.key === "s") { e.preventDefault(); save(); }
});

document.addEventListener("keyup", handler);
document.addEventListener("keypress", handler); // deprecated

Form Events

form.addEventListener("submit", (e) => {
  e.preventDefault();  // stop page reload
  const data = new FormData(form);
  data.get("email");
});

input.addEventListener("input", (e) => {      // fires on every keystroke
  console.log(e.target.value);
});
input.addEventListener("change", handler);    // fires on commit (blur)
input.addEventListener("focus", handler);     // element gains focus
input.addEventListener("blur", handler);      // element loses focus
input.addEventListener("focusin", handler);   // bubbles (unlike focus)
input.addEventListener("focusout", handler);  // bubbles

select.addEventListener("change", (e) => console.log(e.target.value));

Window & Document Events

window.addEventListener("load", handler);           // all resources loaded
document.addEventListener("DOMContentLoaded", handler); // DOM parsed
window.addEventListener("resize", handler);
window.addEventListener("scroll", handler);
window.addEventListener("beforeunload", (e) => {
  e.returnValue = "";   // triggers "Leave site?" dialog
});
window.addEventListener("online", handler);
window.addEventListener("offline", handler);

Event Delegation

// Listen on parent — works for dynamically added children
document.querySelector("#list").addEventListener("click", (e) => {
  const item = e.target.closest("li");  // find the li ancestor
  if (!item) return;
  console.log("Clicked:", item.textContent);
});

Custom Events

// Create and dispatch
const event = new CustomEvent("user:login", {
  detail: { user: "Alice" },
  bubbles: true,
  cancelable: true,
});
document.dispatchEvent(event);

// Listen
document.addEventListener("user:login", (e) => {
  console.log(e.detail.user);  // "Alice"
});
