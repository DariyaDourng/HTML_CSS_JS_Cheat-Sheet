Classes & OOP

ES6 classes, inheritance, static methods, private fields, and prototype patterns.

Class Declaration

class Animal {
  // Public class fields (ES2022)
  species = "Unknown";

  // Private fields (#) — truly private
  #id;
  #energy = 100;

  // Static field
  static count = 0;

  // Constructor
  constructor(name, sound) {
    this.name = name;
    this.sound = sound;
    this.#id = ++Animal.count;
  }

  // Instance method
  speak() {
    return `${this.name} says ${this.sound}!`;
  }

  // Getter
  get id() { return this.#id; }

  // Setter
  set energy(val) {
    if (val < 0) throw new RangeError("Energy cannot be negative");
    this.#energy = val;
  }
  get energy() { return this.#energy; }

  // Static method — called on class, not instance
  static getCount() { return Animal.count; }

  // Static factory method
  static createCat(name) { return new Animal(name, "meow"); }

  // Private method
  #validate() { return this.name.length > 0; }

  // toString
  toString() { return `Animal(${this.name})`; }
}

// Instantiate
const dog = new Animal("Rex", "woof");
dog.speak();            // "Rex says woof!"
dog.id;                  // 1 (via getter)
dog.energy = 80;        // via setter
Animal.getCount();      // 1 (static)
Animal.createCat("Whiskers");

Inheritance with extends

class Dog extends Animal {
  #tricks = [];

  constructor(name, breed) {
    super(name, "woof");  // must call super() first
    this.breed = breed;
  }

  // Override parent method
  speak() {
    return `${super.speak()} (I'm a ${this.breed})`;
  }

  learnTrick(trick) {
    this.#tricks.push(trick);
  }

  perform() {
    return this.#tricks.join(", ") || "No tricks";
  }
}

const d = new Dog("Buddy", "Labrador");
d.speak();           // "Buddy says woof! (I'm a Labrador)"
d.learnTrick("sit");
d instanceof Dog;    // true
d instanceof Animal; // true

Mixins (simulated multiple inheritance)

const Serializable = (Base) => class extends Base {
  serialize() { return JSON.stringify(this); }
  static deserialize(json) { return Object.assign(new this(), JSON.parse(json)); }
};

const Timestamped = (Base) => class extends Base {
  constructor(...args) {
    super(...args);
    this.createdAt = new Date().toISOString();
  }
};

class User extends Timestamped(Serializable(Animal)) {
  constructor(name) { super(name, "hello"); }
}

Prototype-based OOP (pre-class)

// Old-style constructor function
function Person(name, age) {
  this.name = name;
  this.age = age;
}
Person.prototype.greet = function() {
  return `Hi, I'm ${this.name}`;
};

const p = new Person("Alice", 30);
p.greet();  // "Hi, I'm Alice"

// Prototype chain inspection
Object.getPrototypeOf(p) === Person.prototype;  // true
p.hasOwnProperty("name");  // true
p.hasOwnProperty("greet"); // false (on prototype)
