CSS Selectors

Every type of CSS selector — from basic to advanced combinators and pseudo-classes.

Basic Selectors

/* Universal selector */
* { box-sizing: border-box; }

/* Type / element selector */
p  { color: white; }
h1 { font-size: 2rem; }

/* Class selector */
.highlight   { background: yellow; }
.btn.primary { color: white; } /* element with BOTH classes */

/* ID selector (unique per page) */
#header { position: sticky; }

/* Grouping selectors */
h1, h2, h3 { font-family: sans-serif; }

Attribute Selectors

[disabled]             { opacity: 0.5; }        /* has attribute */
[type="text"]          { border: 1px solid; }   /* exact value */
[class~="card"]        { }                      /* space-separated word */
[href^="https"]        { }                      /* starts with */
[href$=".pdf"]         { }                      /* ends with */
[href*="example"]      { }                      /* contains */
[lang|="en"]           { }                      /* en or en-US etc */
[data-theme="dark" i]  { }                      /* case-insensitive */

Combinators

div p     { }  /* Descendant: any p inside div */
div > p   { }  /* Child: direct p child of div */
h2 + p    { }  /* Adjacent sibling: p immediately after h2 */
h2 ~ p    { }  /* General sibling: all p after h2 (same parent) */

Pseudo-Classes

/* User interaction */
a:hover    { }   /* mouse over */
a:focus    { }   /* keyboard focus */
a:active   { }   /* being clicked */
a:visited  { }   /* already visited */
a:link     { }   /* not yet visited */
:focus-visible { }  /* focus via keyboard only */
:focus-within  { }  /* element or descendant has focus */

/* Form states */
input:checked    { }
input:disabled   { }
input:enabled    { }
input:required   { }
input:optional   { }
input:valid      { }
input:invalid    { }
input:placeholder-shown { }
input:read-only  { }

/* Structural */
li:first-child        { }  /* first li */
li:last-child         { }  /* last li */
li:nth-child(2)       { }  /* 2nd li */
li:nth-child(odd)     { }  /* 1, 3, 5... */
li:nth-child(even)    { }  /* 2, 4, 6... */
li:nth-child(3n+1)    { }  /* every 3rd starting at 1 */
li:nth-last-child(1)  { }  /* last item */
p:first-of-type       { }
p:last-of-type        { }
p:nth-of-type(2)      { }
p:only-child          { }
p:only-of-type        { }
:empty                { }  /* no children */
:root                 { }  /* html element */
:not(.active)         { }  /* negation */
:is(h1, h2, h3)       { }  /* matches any in list */
:where(header, footer){ }  /* like :is but zero specificity */
:has(img)             { }  /* parent selector! */

Pseudo-Elements

p::first-line   { }     /* first rendered line */
p::first-letter{ }     /* first letter (drop cap) */
::selection    { }     /* text selected by user */
::placeholder  { }     /* input placeholder text */
::marker       { }     /* list item bullet/number */
::before       { content: ''; }  /* insert before content */
::after        { content: ''; }  /* insert after content */
::backdrop     { }     /* behind dialog/fullscreen */
::file-selector-button { }  /* file input button */
