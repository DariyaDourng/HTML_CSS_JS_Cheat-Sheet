Objects

Object literals, methods, this, prototypes, Object built-ins, Map, Set, and destructuring.

Object Literals

const person = {
  name: "Alice",
  age: 30,
  "full name": "Alice Smith",   // quoted key with space

  // Shorthand property (ES6)
  // if variable name = key name:
  // const age = 30; const obj = { age }; → { age: 30 }

  // Method shorthand
  greet() {
    return `Hi, I'm ${this.name}`;
  },

  // Computed property name
  ["prop_" + "dynamic"]: 42,

  // Getter
  get upperName() {
    return this.name.toUpperCase();
  },

  // Setter
  set fullAge(val) {
    this.age = val;
  },
};

Accessing Properties

person.name;         // dot notation
person["name"];      // bracket notation (needed for dynamic keys)
person["full name"]; // bracket needed for keys with spaces

const key = "age";
person[key];          // dynamic access

// Optional chaining
person?.address?.city;  // undefined (no error if address missing)

Object Built-in Methods

const obj = { a: 1, b: 2, c: 3 };

Object.keys(obj);     // ["a", "b", "c"]
Object.values(obj);   // [1, 2, 3]
Object.entries(obj);  // [["a",1], ["b",2], ["c",3]]

// Create object from entries
Object.fromEntries([["a", 1], ["b", 2]]); // { a:1, b:2 }
Object.fromEntries(new Map([["a", 1]]));  // { a:1 }

// Merge / copy objects
Object.assign({}, obj, { d: 4 });  // { a:1, b:2, c:3, d:4 }
const copy = { ...obj };            // spread (shallow clone)
const merged = { ...obj, d: 4, a: 99 }; // { a:99, b:2, c:3, d:4 }

// Check property exists
"a" in obj;                    // true (incl. inherited)
obj.hasOwnProperty("a");       // true (own only)
Object.hasOwn(obj, "a");       // true (modern, preferred)

// Freeze / seal
Object.freeze(obj);     // no changes allowed (shallow)
Object.seal(obj);       // can update but not add/delete
Object.isFrozen(obj);   // true

// Prototype
Object.getPrototypeOf(obj);
Object.create(null);    // object with no prototype

// Property descriptors
Object.defineProperty(obj, "id", {
  value: 42,
  writable: false,
  enumerable: true,
  configurable: false,
});
Object.getOwnPropertyDescriptor(obj, "id");

Destructuring Objects

const { name, age, city = "Unknown" } = person;

// Rename while destructuring
const { name: fullName, age: years } = person;

// Nested destructuring
const { address: { city, zip } } = person;

// Rest
const { a, ...others } = { a: 1, b: 2, c: 3 };
// a=1, others={b:2, c:3}

// In function params
function render({ title, body, author = "Anon" }) {
  return `${title} by ${author}: ${body}`;
}

Map & Set

// Map — key-value pairs, any type as key
const map = new Map();
map.set("name", "Alice");
map.set(42, "number key");
map.set({}, "object key");
map.get("name");    // "Alice"
map.has("name");    // true
map.delete("name");// removes
map.size;            // number of entries
map.clear();         // removes all
for (const [k, v] of map) console.log(k, v);

// Set — unique values
const set = new Set([1, 2, 2, 3]); // {1, 2, 3}
set.add(4);
set.has(2);    // true
set.delete(1);
set.size;      // 3

// Remove duplicates from array
const unique = [...new Set([1, 2, 2, 3])]; // [1,2,3]

// WeakMap & WeakSet — weakly-held references
const wm = new WeakMap(); // only object keys, not iterable
const ws = new WeakSet(); // only objects, not iterable
