Global Attributes

Attributes that can be applied to any HTML element.

Core Attributes

<!-- id: unique identifier on the page -->
<div id="main-content">...</div>

<!-- class: one or more space-separated class names -->
<p class="text-large text-bold highlight">...</p>

<!-- style: inline CSS (avoid in production) -->
<span style="color: red; font-weight: bold;">Red bold</span>

<!-- title: tooltip on hover -->
<abbr title="As Soon As Possible">ASAP</abbr>

<!-- lang: language of element content -->
<p lang="fr">Bonjour le monde</p>

<!-- dir: text direction -->
<p dir="rtl">Right to left text</p>
<p dir="auto">Auto-detected direction</p>

data-* Custom Attributes

<!-- Store custom data on any element -->
<button
  data-user-id="42"
  data-action="delete"
  data-confirm="Are you sure?"
>Delete</button>

<!-- Access in JavaScript -->
<script>
const btn = document.querySelector('button');
console.log(btn.dataset.userId);   // "42"
console.log(btn.dataset.action);    // "delete"
</script>

Accessibility (ARIA) Attributes

<!-- role: describes element's purpose -->
<div role="button" tabindex="0">Custom button</div>
<nav role="navigation">...</nav>

<!-- aria-label: label when no visible text -->
<button aria-label="Close dialog">✕</button>

<!-- aria-labelledby: points to another element -->
<h2 id="dialog-title">Login</h2>
<div aria-labelledby="dialog-title">...</div>

<!-- aria-hidden: hide from screen readers -->
<span aria-hidden="true">🎉</span>

<!-- aria-expanded: open/closed state -->
<button aria-expanded="false" aria-controls="menu">Toggle</button>

<!-- aria-live: announce dynamic updates -->
<div aria-live="polite">Status updates here</div>

<!-- aria-required + aria-invalid for forms -->
<input type="email" aria-required="true" aria-invalid="false"/>

Interactive Global Attributes

<!-- tabindex: keyboard focus order -->
<div tabindex="0">Focusable</div>     <!-- in tab order -->
<div tabindex="-1">Focusable via JS</div> <!-- not in tab order -->

<!-- hidden: hide element -->
<p hidden>Not visible</p>

<!-- contenteditable: make any element editable -->
<div contenteditable="true">Click to edit this text!</div>

<!-- draggable -->
<img src="icon.png" draggable="true" alt="Drag me"/>

<!-- spellcheck -->
<textarea spellcheck="true"></textarea>

<!-- translate: should translators translate this? -->
<span translate="no">Brand Name</span>

Event Handler Attributes (inline — avoid in production)

<button onclick="alert('clicked')">Click</button>
<input oninput="console.log(this.value)"/>
<img onload="console.log('loaded')" onerror="this.src='fallback.png'" src="img.jpg" alt="img"/>
<!-- Better practice: addEventListener() in JavaScript -->
