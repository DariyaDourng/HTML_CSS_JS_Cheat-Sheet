Error Handling

try/catch/finally, throw, all Error types, custom errors, and debugging.

try / catch / finally

try {
  // Code that may throw
  const data = JSON.parse("invalid json");
} catch (error) {
  // Runs when error is thrown
  console.error(error.name);     // "SyntaxError"
  console.error(error.message);  // "Unexpected token i..."
  console.error(error.stack);    // Full stack trace
} finally {
  // Always runs — even if no error, even after return
  console.log("Cleanup done");
}

// Without catch (just ensure cleanup)
try {
  openConnection();
} finally {
  closeConnection();
}

throw

// Can throw anything
throw "simple string";
throw 42;
throw { code: 404, message: "Not found" };

// Best practice — throw Error objects
throw new Error("Something went wrong");
throw new TypeError("Expected a string");

// Guard clauses
function divide(a, b) {
  if (typeof a !== "number") throw new TypeError("a must be a number");
  if (b === 0) throw new RangeError("Cannot divide by zero");
  return a / b;
}

Custom Error Classes

class ValidationError extends Error {
  constructor(message, field) {
    super(message);
    this.name = "ValidationError";
    this.field = field;
  }
}

class HttpError extends Error {
  constructor(status, message) {
    super(message);
    this.name = "HttpError";
    this.status = status;
  }
}

// Usage
try {
  throw new ValidationError("Name is required", "name");
} catch (e) {
  if (e instanceof ValidationError) {
    console.log(`Field: ${e.field}`);
  } else if (e instanceof HttpError) {
    console.log(`Status: ${e.status}`);
  } else {
    throw e;  // rethrow unknown errors
  }
}

Async Error Handling

// async/await — use try/catch
async function fetchUser(id) {
  try {
    const res = await fetch(`/api/users/${id}`);
    if (!res.ok) throw new HttpError(res.status, res.statusText);
    return await res.json();
  } catch (err) {
    if (err instanceof HttpError && err.status === 404) {
      return null;   // recover gracefully
    }
    throw err;       // rethrow others
  }
}

// Global handlers
window.addEventListener("error", (e) => {
  console.error("Uncaught:", e.error);
});
window.addEventListener("unhandledrejection", (e) => {
  console.error("Unhandled Promise rejection:", e.reason);
  e.preventDefault();  // prevent default browser logging
});

Debugging Tips

console.log("value:", x);
console.error("Error!", err);
console.warn("Warning");
console.info("Info");
console.table(arrayOrObject);    // pretty table
console.dir(element);            // DOM element properties
console.group("Group"); console.groupEnd();

console.count("label");   // count calls
console.assert(x > 0, "x must be positive");
console.trace();           // print stack trace

debugger;  // pause execution in DevTools
