Variables & Data Types

var, let, const, all primitive types, typeof, and type coercion.

Variable Declarations

// var — function-scoped, hoisted, re-declarable (avoid)
var name = "Alice";
var name = "Bob";   // OK (re-declaration)

// let — block-scoped, not hoisted (use for mutable values)
let score = 0;
score = 10;          // OK (re-assignment)
// let score = 5;    // Error: already declared

// const — block-scoped, must be initialized, cannot reassign
const PI = 3.14159;
// PI = 3;           // Error: assignment to constant
const user = { name: "Alice" };
user.name = "Bob"; // OK — object contents can still change

Primitive Data Types

// 1. String
let a = "double quotes";
let b = 'single quotes';
let c = `template literal: ${a}`;  // backticks, allows interpolation
let multi = `Line 1
Line 2`;  // multi-line

// 2. Number (64-bit float — covers integers too)
let int   = 42;
let float = 3.14;
let neg   = -7;
let hex   = 0xFF;       // 255
let bin   = 0b1010;    // 10
let oct   = 0o17;      // 15
let exp   = 2e6;       // 2000000
let sep   = 1_000_000; // numeric separator (ES2021)

// Special number values
Infinity;   // 1/0
-Infinity;
NaN;        // Not a Number — result of invalid math
Number.MAX_SAFE_INTEGER;  // 9007199254740991
Number.MIN_SAFE_INTEGER;

// 3. Boolean
let isOn  = true;
let isOff = false;

// 4. null — intentional absence of value
let empty = null;

// 5. undefined — variable declared but not assigned
let x;
console.log(x);  // undefined

// 6. Symbol — unique identifier
const sym1 = Symbol("id");
const sym2 = Symbol("id");
sym1 === sym2;  // false — always unique
Symbol.for("shared");  // global symbol registry

// 7. BigInt — arbitrarily large integers
const big = 9007199254740993n;
const big2 = BigInt(9007199254740993);

typeof Operator

typeof "hello"     // "string"
typeof 42          // "number"
typeof 3.14        // "number"
typeof NaN         // "number" (quirk!)
typeof true        // "boolean"
typeof undefined   // "undefined"
typeof null        // "object" (famous bug — remains for compatibility)
typeof Symbol()    // "symbol"
typeof 42n         // "bigint"
typeof {}          // "object"
typeof []          // "object" (arrays are objects!)
typeof function(){} // "function"

Type Conversion

// To String
String(42);          // "42"
String(true);       // "true"
String(null);       // "null"
(42).toString();    // "42"
(255).toString(16); // "ff" (hex)
(8).toString(2);   // "1000" (binary)

// To Number
Number("42");       // 42
Number("");         // 0
Number(true);       // 1
Number(false);      // 0
Number(null);       // 0
Number(undefined);  // NaN
Number("abc");      // NaN
parseInt("42px");   // 42 (stops at non-digit)
parseFloat("3.14m");// 3.14
+"42";               // 42 (unary plus)

// To Boolean
Boolean(0);         // false
Boolean("");        // false
Boolean(null);      // false
Boolean(undefined); // false
Boolean(NaN);       // false
// Everything else is truthy (including "0", [], {})

// Falsy values (only 6):
false, 0, -0, 0n, "", null, undefined, NaN

Variable Scope & Hoisting

// var is function-scoped and hoisted
greet();  // works — function hoisted
console.log(x); // undefined — var declaration hoisted, not value
var x = 5;

// let/const are block-scoped and NOT hoisted (Temporal Dead Zone)
{
  let inner = "only here";
  const also = "only here";
}
// console.log(inner); // ReferenceError

function greet() { return "hello"; }
