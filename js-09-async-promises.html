Async & Promises

Callbacks, Promises, async/await, fetch API, and error handling.

Callbacks (old style)

// Callback-based async
setTimeout(() => console.log("1 second later"), 1000);
setInterval(() => console.log("every second"), 1000);

// Callback hell (the problem)
getData((data) => {
  processData(data, (result) => {
    saveResult(result, (saved) => {
      console.log("done");  // hard to read and handle errors
    });
  });
});

Promise

// Creating a Promise
const p = new Promise((resolve, reject) => {
  const success = true;
  if (success) {
    resolve("Data loaded!");
  } else {
    reject(new Error("Failed"));
  }
});

// Consuming a Promise
p
  .then((value) => {
    console.log(value);   // "Data loaded!"
    return "Next";       // can chain
  })
  .then((value) => {
    console.log(value);   // "Next"
  })
  .catch((error) => {
    console.log(error);   // handles any rejection in chain
  })
  .finally(() => {
    console.log("Always runs");
  });

Promise Static Methods

// All succeed or fail fast
Promise.all([p1, p2, p3])
  .then(([r1, r2, r3]) => console.log(r1, r2, r3));

// All settle (never rejects)
Promise.allSettled([p1, p2]).then((results) => {
  results.forEach(({ status, value, reason }) => {
    if (status === "fulfilled") console.log(value);
    else console.log(reason);
  });
});

// First to resolve
Promise.race([p1, p2]).then((winner) => console.log(winner));

// First to succeed (ignores rejections)
Promise.any([p1, p2]).then((first) => console.log(first));

// Already resolved/rejected
Promise.resolve("value");
Promise.reject(new Error("oops"));

async / await

// async functions always return a Promise
async function loadUser(id) {
  const response = await fetch(`/api/users/${id}`);

  if (!response.ok) {
    throw new Error(`HTTP ${response.status}`);
  }

  const user = await response.json();
  return user;
}

// Call it
loadUser(42)
  .then((user) => console.log(user))
  .catch((err) => console.error(err));

// Or with async/await at the call site
async function main() {
  try {
    const user = await loadUser(42);
    console.log(user);
  } catch (err) {
    console.error(err);
  }
}

Parallel vs Sequential

// Sequential — one after another (slower)
const a = await fetchA();
const b = await fetchB();

// Parallel — run at same time (faster)
const [a, b] = await Promise.all([fetchA(), fetchB()]);

// Parallel with error handling per item
const results = await Promise.allSettled([fetchA(), fetchB()]);

fetch API

// GET request
const res = await fetch("https://api.example.com/data");
const json = await res.json();
// res.text(), res.blob(), res.arrayBuffer(), res.formData()

// POST request
const res = await fetch("/api/users", {
  method: "POST",
  headers: { "Content-Type": "application/json" },
  body: JSON.stringify({ name: "Alice" }),
});

// Check for errors (fetch doesn't throw on HTTP errors!)
if (!res.ok) throw new Error(`HTTP ${res.status}: ${res.statusText}`);

// Abort a fetch
const controller = new AbortController();
fetch("/api", { signal: controller.signal });
controller.abort();   // cancel request

// Timeout pattern
const timeout = (ms) => new Promise((_, reject) =>
  setTimeout(() => reject(new Error("Timeout")), ms)
);
const data = await Promise.race([fetch("/api"), timeout(5000)]);

Async Iteration

// for await...of
async function processStream() {
  const response = await fetch("/api/stream");
  for await (const chunk of response.body) {
    console.log(chunk);
  }
}

// Async generator
async function* paginate(url) {
  while (url) {
    const res = await fetch(url);
    const data = await res.json();
    yield data.items;
    url = data.nextPage;
  }
}
