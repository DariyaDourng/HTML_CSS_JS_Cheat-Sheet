Forms & Inputs

All form elements, every input type, validation attributes, and best practices.

Form Element

<form
  action="/submit"      <!-- Where to send data -->
  method="post"        <!-- GET or POST -->
  enctype="multipart/form-data"  <!-- Required for file uploads -->
  novalidate           <!-- Disable browser validation -->
  autocomplete="on"   <!-- Enable/disable autocomplete -->
>
  <!-- form fields -->
</form>

All Input Types

<!-- Text inputs -->
<input type="text"          placeholder="Plain text"/>
<input type="email"         placeholder="user@email.com"/>
<input type="password"      placeholder="••••••••"/>
<input type="search"        placeholder="Search..."/>
<input type="url"           placeholder="https://..."/>
<input type="tel"           placeholder="+1 555 000 0000"/>

<!-- Number inputs -->
<input type="number"  min="0" max="100" step="1"/>
<input type="range"   min="0" max="100" value="50"/>

<!-- Date/time inputs -->
<input type="date"/>
<input type="time"/>
<input type="datetime-local"/>
<input type="month"/>
<input type="week"/>

<!-- Other -->
<input type="checkbox" checked/>
<input type="radio"    name="group" value="a"/>
<input type="color"   value="#00ffd1"/>
<input type="file"    accept="image/*" multiple/>
<input type="hidden"  name="token" value="abc123"/>
<input type="submit"  value="Submit"/>
<input type="reset"   value="Clear"/>
<input type="button"  value="Click Me"/>
<input type="image"   src="btn.png" alt="Submit"/>

Label, Textarea, Select

<!-- Label linked by for/id -->
<label for="username">Username:</label>
<input type="text" id="username" name="username"/>

<!-- Textarea -->
<textarea name="message" rows="5" cols="40"
           placeholder="Your message..."></textarea>

<!-- Select dropdown -->
<select name="country">
  <option value="">-- Select country --</option>
  <optgroup label="Asia">
    <option value="kh">Cambodia</option>
    <option value="th" selected>Thailand</option>
  </optgroup>
</select>

<!-- Multi-select -->
<select name="skills" multiple size="4">
  <option value="html">HTML</option>
  <option value="css">CSS</option>
  <option value="js">JavaScript</option>
</select>

Validation Attributes

<input type="text"
  required                        <!-- Must be filled -->
  minlength="3"                  <!-- Min characters -->
  maxlength="50"                 <!-- Max characters -->
  pattern="[A-Za-z]{3,}"        <!-- Regex pattern -->
  title="3+ letters only"        <!-- Error message hint -->
/>

<input type="number"
  min="18" max="99"
  step="1"
  required
/>

Fieldset & Button Types

<fieldset>
  <legend>Personal Information</legend>
  <label>Name: <input type="text"/></label>
</fieldset>

<!-- Button types -->
<button type="submit">Submit form</button>
<button type="reset">Reset form</button>
<button type="button" onclick="doSomething()">Click</button>
<button disabled>Disabled</button>

<!-- Datalist for autocomplete suggestions -->
<input list="browsers" name="browser" placeholder="Pick a browser"/>
<datalist id="browsers">
  <option value="Chrome"/>
  <option value="Firefox"/>
  <option value="Safari"/>
</datalist>
