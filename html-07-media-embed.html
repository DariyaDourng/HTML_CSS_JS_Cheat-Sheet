Media & Embeds

Video, audio, iframes, canvas, inline SVG, and embedded content.

Video Element

<!-- Basic video -->
<video src="movie.mp4" controls>
  Your browser does not support the video element.
</video>

<!-- Full-featured video -->
<video
  controls                       <!-- Show play/pause/volume -->
  autoplay                       <!-- Autoplay (needs muted on most browsers) -->
  muted                          <!-- Start muted -->
  loop                           <!-- Loop forever -->
  playsinline                    <!-- iOS: don't fullscreen automatically -->
  poster="thumbnail.jpg"         <!-- Thumbnail before play -->
  width="640" height="360"
  preload="metadata"            <!-- none | metadata | auto -->
>
  <source src="movie.webm" type="video/webm"/>
  <source src="movie.mp4"  type="video/mp4"/>
  <track src="subs.vtt" kind="subtitles" srclang="en" label="English" default/>
</video>

Audio Element

<audio controls>
  <source src="music.ogg" type="audio/ogg"/>
  <source src="music.mp3" type="audio/mpeg"/>
  <source src="music.wav" type="audio/wav"/>
</audio>

IFrame — Embedding External Content

<!-- Basic iframe -->
<iframe
  src="https://example.com"
  width="600"
  height="400"
  title="Embedded content"       <!-- Required for accessibility -->
  loading="lazy"
  sandbox="allow-scripts allow-same-origin"
></iframe>

<!-- YouTube embed -->
<iframe
  src="https://www.youtube.com/embed/VIDEO_ID"
  width="560" height="315"
  title="YouTube video"
  allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture"
  allowfullscreen
></iframe>

<!-- Google Maps embed -->
<iframe
  src="https://www.google.com/maps/embed?pb=..."
  width="600" height="450"
  style="border:0"
  allowfullscreen
  loading="lazy"
  referrerpolicy="no-referrer-when-downgrade"
></iframe>

Canvas Element

<!-- Canvas: drawn via JavaScript -->
<canvas id="myCanvas" width="400" height="200">
  Fallback text for non-supporting browsers.
</canvas>

<script>
const canvas = document.getElementById('myCanvas');
const ctx = canvas.getContext('2d');
ctx.fillStyle = '#00ffd1';
ctx.fillRect(10, 10, 100, 80);
</script>

Inline SVG

<!-- SVG embedded directly in HTML -->
<svg xmlns="http://www.w3.org/2000/svg"
     width="100" height="100"
     viewBox="0 0 100 100"
     aria-label="Green circle"
     role="img">
  <circle cx="50" cy="50" r="40" fill="#00ffd1"/>
  <text x="50" y="55" text-anchor="middle" fill="black">SVG</text>
</svg>
