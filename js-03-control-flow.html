Control Flow

if/else, switch, all loop types, break, continue, and labels.

if / else if / else

if (score >= 90) {
  console.log("A");
} else if (score >= 80) {
  console.log("B");
} else if (score >= 70) {
  console.log("C");
} else {
  console.log("Fail");
}

// Single-line (no braces — use cautiously)
if (x > 0) console.log("positive");

// Ternary shorthand
const label = x > 0 ? "pos" : "neg";

switch / case

switch (day) {
  case "Monday":
  case "Tuesday":
    console.log("Weekday");
    break;          // stop — without break, falls through!

  case "Saturday":
  case "Sunday":
    console.log("Weekend");
    break;

  default:          // runs if no case matches
    console.log("Unknown");
}

// switch uses strict equality (===)

for Loop

// Classic for
for (let i = 0; i < 5; i++) {
  console.log(i);  // 0, 1, 2, 3, 4
}

// Counting down
for (let i = 10; i > 0; i--) {
  console.log(i);
}

// Step by 2
for (let i = 0; i <= 10; i += 2) {
  console.log(i);  // 0, 2, 4, 6, 8, 10
}

// Iterate an array (classic)
const fruits = ["apple", "banana", "cherry"];
for (let i = 0; i < fruits.length; i++) {
  console.log(fruits[i]);
}

for...of — Iterate Values

// Arrays
for (const fruit of fruits) {
  console.log(fruit);
}

// With index using entries()
for (const [index, fruit] of fruits.entries()) {
  console.log(index, fruit);
}

// Strings
for (const char of "hello") {
  console.log(char);  // h e l l o
}

// Maps and Sets
const map = new Map([["a", 1], ["b", 2]]);
for (const [key, val] of map) {
  console.log(key, val);
}

// NodeList (DOM elements)
for (const el of document.querySelectorAll("p")) {
  el.style.color = "red";
}

for...in — Iterate Object Keys

const person = { name: "Alice", age: 30, city: "Phnom Penh" };

for (const key in person) {
  console.log(key, person[key]);
}
// name Alice
// age 30
// city Phnom Penh

// hasOwnProperty check (excludes inherited)
for (const key in person) {
  if (person.hasOwnProperty(key)) {
    console.log(key);
  }
}

while Loop

let count = 0;
while (count < 5) {
  console.log(count);
  count++;
}

// Infinite loop with break
while (true) {
  const input = getInput();
  if (input === "quit") break;
  processInput(input);
}

do...while Loop

// Always runs at least once
let n = 0;
do {
  console.log(n);
  n++;
} while (n < 3);

break, continue, Labels

// break — exit loop immediately
for (let i = 0; i < 10; i++) {
  if (i === 5) break;
  console.log(i);  // 0 1 2 3 4
}

// continue — skip current iteration
for (let i = 0; i < 5; i++) {
  if (i === 2) continue;
  console.log(i);  // 0 1 3 4
}

// Labeled break — break outer loop from inner
outer: for (let i = 0; i < 3; i++) {
  for (let j = 0; j < 3; j++) {
    if (j === 1) break outer;
    console.log(i, j);
  }
}
