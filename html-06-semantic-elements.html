Semantic Elements

HTML5 semantic tags that describe the meaning of content for browsers and screen readers.

Page Layout Structure

<body>

  <!-- Site header / banner -->
  <header>
    <nav>  <!-- Primary navigation -->
      <ul>
        <li><a href="/">Home</a></li>
        <li><a href="/about">About</a></li>
      </ul>
    </nav>
  </header>

  <!-- Main content area (one per page) -->
  <main>

    <!-- Standalone thematic section -->
    <section>
      <h2>Featured Posts</h2>

      <!-- Self-contained, independently distributable content -->
      <article>
        <header>
          <h3>Article Title</h3>
          <time datetime="2024-11-01">November 1, 2024</time>
        </header>
        <p>Article content...</p>
        <footer>
          <address>By <a href="mailto:a@b.com">Author</a></address>
        </footer>
      </article>

    </section>

    <!-- Tangentially related content -->
    <aside>
      <h3>Related Links</h3>
    </aside>

  </main>

  <!-- Site footer -->
  <footer>
    <p>&copy; 2024 My Company</p>
  </footer>

</body>

Interactive Elements

<!-- details + summary: accordion / disclosure widget -->
<details>
  <summary>Click to expand</summary>
  <p>Hidden content revealed on click.</p>
</details>

<!-- Open by default -->
<details open>
  <summary>Already expanded</summary>
  <p>This is visible by default.</p>
</details>

<!-- dialog element -->
<dialog id="myDialog">
  <p>This is a native dialog.</p>
  <button onclick="myDialog.close()">Close</button>
</dialog>
<button onclick="document.getElementById('myDialog').showModal()">Open Dialog</button>

<!-- progress and meter -->
<progress value="70" max="100">70%</progress>
<meter value="0.6" min="0" max="1" low="0.25" high="0.75" optimum="1">60%</meter>

Other Semantic Inline Elements

<!-- time: machine-readable date/time -->
<time datetime="2024-12-25T00:00">Christmas Day</time>

<!-- data: machine-readable value -->
<data value="42">Forty-two</data>

<!-- address: contact information -->
<address>
  123 Main Street, Phnom Penh, Cambodia<br>
