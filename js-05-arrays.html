Arrays

Create, read, mutate, iterate, and transform arrays — every built-in method.

Creating Arrays

const arr1 = [1, 2, 3];
const arr2 = new Array(3);          // [empty × 3]
const arr3 = new Array(1, 2, 3);    // [1, 2, 3]
const arr4 = Array.from({ length: 5 }, (_, i) => i + 1); // [1,2,3,4,5]
const arr5 = Array.from("hello");    // ['h','e','l','l','o']
const arr6 = Array.from(new Set([1, 2, 2, 3])); // [1,2,3]
const arr7 = Array.of(1, 2, 3);     // [1, 2, 3]
const arr8 = [0, ...[1, 2], 3];    // spread: [0,1,2,3]

Reading & Basic Operations

const a = ["a", "b", "c", "d"];
a[0];           // "a"
a[a.length - 1]; // "d" — last element
a.at(-1);        // "d" — modern, negative index
a.length;        // 4
Array.isArray(a);// true

Mutating Methods (change original)

const a = [1, 2, 3];

// Add / remove at end
a.push(4, 5);       // [1,2,3,4,5] — returns new length
a.pop();            // removes & returns 5 → [1,2,3,4]

// Add / remove at start
a.unshift(0);       // [0,1,2,3,4] — returns new length
a.shift();          // removes & returns 0 → [1,2,3,4]

// splice(start, deleteCount, ...items)
a.splice(1, 0, 99);  // insert at index 1: [1,99,2,3,4]
a.splice(2, 1);      // delete 1 at index 2: [1,99,3,4]
a.splice(1, 2, 10, 20); // replace 2 at 1: [1,10,20,4]

// Sort
[3, 1, 2].sort();              // [1,2,3] — lexicographic default!
[10, 9, 2].sort((a, b) => a - b); // ascending [2,9,10]
[10, 9, 2].sort((a, b) => b - a); // descending [10,9,2]

// Reverse
[1, 2, 3].reverse();  // [3,2,1]

// Fill
new Array(5).fill(0);         // [0,0,0,0,0]
[1,2,3,4].fill(0, 1, 3);    // [1,0,0,4]

// copyWithin(target, start, end)
[1,2,3,4,5].copyWithin(0, 3); // [4,5,3,4,5]

Non-Mutating Methods (return new array)

const nums = [1, 2, 3, 4, 5];

// map — transform each element
nums.map(n => n * 2);             // [2,4,6,8,10]

// filter — keep matching elements
nums.filter(n => n % 2 === 0);    // [2,4]

// reduce(callback, initialValue)
nums.reduce((acc, n) => acc + n, 0); // 15
nums.reduce((acc, n) => ({ ...acc, [n]: n * n }), {}); // to object

// reduceRight — right to left
[[1], [2], [3]].reduceRight((acc, a) => acc.concat(a), []); // [3,2,1]

// slice(start, end) — extract subarray
nums.slice(1, 3);   // [2,3]
nums.slice(-2);     // [4,5]

// concat — merge arrays
[1, 2].concat([3, 4], [5]); // [1,2,3,4,5]

// flat & flatMap
[1, [2, [3]]].flat();           // [1,2,[3]] — depth 1
[1, [2, [3]]].flat(Infinity);   // [1,2,3]
nums.flatMap(n => [n, n * 2]);  // [1,2,2,4,3,6,4,8,5,10]

// toSorted, toReversed, with (non-mutating, ES2023)
nums.toSorted((a, b) => b - a); // new sorted array
nums.toReversed();               // new reversed array
nums.with(2, 99);                // [1,2,99,4,5] — replace at index

Search Methods

const a = [10, 20, 30, 20];

a.indexOf(20);          // 1  (first occurrence)
a.lastIndexOf(20);      // 3  (last occurrence)
a.includes(30);         // true
a.find(n => n > 15);    // 20 (first match)
a.findLast(n => n > 15);// 20 (last match)
a.findIndex(n => n > 15); // 1
a.findLastIndex(n => n > 15); // 3
a.some(n => n > 25);    // true (any match?)
a.every(n => n > 5);    // true (all match?)

Iteration Methods

["a", "b"].forEach((val, idx) => console.log(idx, val));

// entries, keys, values — return iterators
for (const [i, v] of ["a", "b"].entries()) console.log(i, v);
for (const k of ["a", "b"].keys()) console.log(k);
for (const v of ["a", "b"].values()) console.log(v);

String Conversion

[1, 2, 3].join(", ");  // "1, 2, 3"
[1, 2, 3].join("");    // "123"
[1, 2, 3].toString();  // "1,2,3"

Destructuring

const [first, second, ...rest] = [1, 2, 3, 4, 5];
// first=1, second=2, rest=[3,4,5]

// Skip elements
const [, b, , d] = [1, 2, 3, 4]; // b=2, d=4

// Default values
const [x = 10, y = 20] = [5];   // x=5, y=20

// Swap variables
let a = 1, b = 2;
[a, b] = [b, a];  // a=2, b=1
