Lists & Tables

Ordered, unordered, and definition lists; full table syntax with accessibility.

Unordered List

<ul>
  <li>Apples</li>
  <li>Bananas</li>
  <li>Cherries</li>
</ul>

Ordered List

<!-- Default: 1, 2, 3... -->
<ol>
  <li>First step</li>
  <li>Second step</li>
  <li>Third step</li>
</ol>

<!-- start: begin from a specific number -->
<ol start="5">
  <li>Item five</li>
  <li>Item six</li>
</ol>

<!-- reversed: count down -->
<ol reversed>
  <li>Gold</li>
  <li>Silver</li>
  <li>Bronze</li>
</ol>

<!-- type attribute -->
<ol type="A"><li>A</li><li>B</li></ol>  <!-- A, B, C -->
<ol type="a"><li>a</li><li>b</li></ol>  <!-- a, b, c -->
<ol type="I"><li>I</li><li>II</li></ol> <!-- Roman numerals -->

Full Table

<table>
  <!-- caption: accessible table title -->
  <caption>Monthly Sales Data</caption>

  <!-- Column grouping for styling -->
  <colgroup>
    <col span="1" style="background:#111"/>
    <col span="2"/>
  </colgroup>

  <!-- Table head -->
  <thead>
    <tr>
      <th scope="col">Month</th>
      <th scope="col">Sales</th>
      <th scope="col">Revenue</th>
    </tr>
  </thead>

  <!-- Table body -->
  <tbody>
    <tr>
      <th scope="row">January</th>
      <td>120</td>
      <td>$6,000</td>
    </tr>
    <tr>
      <th scope="row">February</th>
      <td>98</td>
      <td>$4,900</td>
    </tr>
  </tbody>

  <!-- Table footer -->
  <tfoot>
    <tr>
      <th>Total</th>
      <td>218</td>
      <td>$10,900</td>
    </tr>
  </tfoot>
</table>

colspan & rowspan

<table>
  <tr>
    <td colspan="2">Spans 2 columns</td>
  </tr>
  <tr>
    <td rowspan="2">Spans 2 rows</td>
    <td>Cell A</td>
  </tr>
  <tr>
    <td>Cell B</td>
  </tr>
</table>
